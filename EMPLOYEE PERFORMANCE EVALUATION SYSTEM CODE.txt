CREATE DATABASE EmployeeEvaluation;
USE EmployeeEvaluation;

CREATE TABLE Department (
 Department_ID INT PRIMARY KEY,
 Department_Name VARCHAR(50) NOT NULL
);
CREATE TABLE Employee (
 Employee_ID INT PRIMARY KEY,
 Employee_Name VARCHAR(50) NOT NULL,
 Department_ID INT,
 Designation VARCHAR(50),
 Date_Of_Joining DATE,
 FOREIGN KEY (Department_ID) REFERENCES Department(Department_ID)
);
CREATE TABLE Evaluator (
 Evaluator_ID INT PRIMARY KEY,
 Evaluator_Name VARCHAR(50) NOT NULL,
 Role VARCHAR(50)
);
CREATE TABLE Evaluation (
 Evaluation_ID INT PRIMARY KEY,
 Employee_ID INT,
 Evaluator_ID INT,
 Performance_Score INT CHECK (Performance_Score BETWEEN 0 AND 100),
 Remarks VARCHAR(100),
 Evaluation_Date DATE,
 FOREIGN KEY (Employee_ID) REFERENCES Employee(Employee_ID),
 FOREIGN KEY (Evaluator_ID) REFERENCES Evaluator(Evaluator_ID)
);

SELECT Employee_Name, Performance_Score
FROM Employee
JOIN Evaluation ON Employee.Employee_ID = Evaluation.Employee_ID;

SELECT Department_ID, COUNT(Employee_ID) AS Total_Employees
FROM Employee
GROUP BY Department_ID;
SELECT Department_ID, COUNT(Employee_ID) AS Total_Employees
FROM Employee
GROUP BY Department_ID
HAVING COUNT(Employee_ID) > 1;

SELECT Employee_Name
FROM Employee
WHERE Employee_ID IN (
 SELECT Employee_ID
 FROM Evaluation
 WHERE Performance_Score > 80
);

CREATE VIEW Performance_View AS
SELECT Employee_Name, Performance_Score
FROM Employee
JOIN Evaluation ON Employee.Employee_ID = Evaluation.Employee_ID;
